  :KUBECONFIG = "kubeconfig-local.yaml"
  kubectl get nodes
  
  # Option 2: Using bridge network (better for production-like testing)
  :KUBECONFIG = "kubeconfig-bridge.yaml"
  kubectl get nodes

To SSH into VM:
  vagrant ssh

To check cluster inside VM:
  vagrant ssh
  kubectl get nodes
  kubectl get pods --all-namespaces

To stop VM:
  vagrant halt

To start VM again:
  vagrant up

To destroy VM (delete everything):
  vagrant destroy
