# Source: margin/charts/sv/templates/optional-services/kafka.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: margin-service-virtualization-kafka-events
  namespace: margin
  labels:
    helm.sh/chart: sv-1.0.0
    app.kubernetes.io/name: service-virtualization
    app.kubernetes.io/instance: margin
    app.kubernetes.io/version: "1.0.0"
    app.kubernetes.io/managed-by: Helm
    app.kubernetes.io/component: kafka
    app.kubernetes.io/instance-name: events
spec:
  replicas: 1
  selector:
    matchLabels:
      app.kubernetes.io/name: service-virtualization
      app.kubernetes.io/instance: margin
      app.kubernetes.io/component: kafka
      app.kubernetes.io/instance-name: events
  template:
    metadata:
      labels:
        app.kubernetes.io/name: service-virtualization
        app.kubernetes.io/instance: margin
        app.kubernetes.io/component: kafka
        app.kubernetes.io/instance-name: events
    spec:
      serviceAccountName: margin-sa
      initContainers:
      # -----------------------------------------------------------------------
      # cert-converter: PEM ΓåÆ JKS using the same cp-kafka image (already pulled).
      # The Confluent cp-kafka image includes both openssl and keytool, so no
      # extra image is needed. This follows the Confluent-standard approach of
      # placing JKS files + credential files in /etc/kafka/secrets.
      # -----------------------------------------------------------------------
      - name: cert-converter
        image: "confluentinc/cp-kafka:7.4.0"
        imagePullPolicy: IfNotPresent
        command:
          - /bin/sh
          - -c
          - |
            set -e
            mkdir -p /etc/kafka/secrets

            echo "Converting PEM certificate to PKCS12..."
            openssl pkcs12 -export \
              -in /mnt/wildcard/tls.crt \
              -inkey /mnt/wildcard/tls.key \
              -out /tmp/kafka.p12 \
              -passout pass:changeit \
              -name kafka

            echo "Converting PKCS12 to JKS keystore..."
            keytool -importkeystore \
              -srckeystore /tmp/kafka.p12 \
              -srcstoretype PKCS12 \
              -srcstorepass changeit \
              -destkeystore /etc/kafka/secrets/kafka.keystore.jks \
              -deststorepass changeit \
              -noprompt

            echo "Creating JKS truststore from server certificate..."
            keytool -importcert \
              -alias wildcard \
              -file /mnt/wildcard/tls.crt \
              -keystore /etc/kafka/secrets/kafka.truststore.jks \
              -storepass changeit \
              -noprompt

            echo "Writing Confluent credential files..."
            echo "changeit" > /etc/kafka/secrets/keystore_credentials
            echo "changeit" > /etc/kafka/secrets/key_credentials
            echo "changeit" > /etc/kafka/secrets/truststore_credentials

            echo "Done. JKS files:"
            ls -lh /etc/kafka/secrets/
        volumeMounts:
          - name: kafka-secrets-events
            mountPath: /etc/kafka/secrets
          - name: margin-tls-secret
            mountPath: /mnt/wildcard
            readOnly: true

      containers:
      - name: kafka
        image: "confluentinc/cp-kafka:7.4.0"
        imagePullPolicy: IfNotPresent
        ports:
        - containerPort: 9092
          name: plaintext
        - containerPort: 9093
          name: controller
        - containerPort: 9094
          name: ssl
        env:
        # =======================================================================
        # KRaft Mode (no Zookeeper)
        # =======================================================================
        # This broker also acts as controller ΓÇö single-node setup for service virt.
        - name: KAFKA_PROCESS_ROLES
          value: "broker,controller"
        - name: KAFKA_NODE_ID
          value: "1"
        - name: KAFKA_CONTROLLER_QUORUM_VOTERS
          value: "1@margin-service-virtualization-kafka-events:9093"
        - name: KAFKA_CONTROLLER_LISTENER_NAMES
          value: "CONTROLLER"
        - name: KAFKA_LOG_DIRS
          value: "/var/lib/kafka/data"
        # KRaft cluster ID ΓÇö must be a valid 22-char base64-encoded UUID.
        # Generate a new one with: kafka-storage.sh random-uuid
        - name: CLUSTER_ID
          value: "MkU3OEVBNTcwNTJENDM2Qk"
        # =======================================================================
        # Listener Configuration
        # Internal: PLAINTEXT://9092  (backend ΓåÆ no TLS needed inside cluster)
        # Controller: CONTROLLER://9093  (KRaft, always internal)
        # External: SSL://9094 (Istio TLS PASSTHROUGH ΓåÆ pod:9094)
        # =======================================================================
        - name: KAFKA_LISTENERS
          value: "PLAINTEXT://0.0.0.0:9092,SSL://0.0.0.0:9094,CONTROLLER://0.0.0.0:9093"
        - name: KAFKA_ADVERTISED_LISTENERS
          # SSL listener advertises the external hostname so clients that reconnect
          # after bootstrap still route through Istio and not directly to the pod.
          value: "PLAINTEXT://margin-service-virtualization-kafka-events.margin.svc.cluster.local:9092,SSL://events.kafka.margin.service-virtualization.local:30501"
        - name: KAFKA_LISTENER_SECURITY_PROTOCOL_MAP
          value: "CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT,SSL:SSL"
        - name: KAFKA_INTER_BROKER_LISTENER_NAME
          value: "PLAINTEXT"
        # =======================================================================
        # SSL Configuration (Confluent convention)
        # Files are placed in /etc/kafka/secrets/ by the cert-converter init container.
        # KAFKA_SSL_KEYSTORE_FILENAME / TRUSTSTORE_FILENAME are relative filenames;
        # the Confluent entrypoint prepends /etc/kafka/secrets/ automatically.
        # =======================================================================
        - name: KAFKA_SSL_KEYSTORE_FILENAME
          value: "kafka.keystore.jks"
        - name: KAFKA_SSL_KEYSTORE_CREDENTIALS
          value: "keystore_credentials"
        - name: KAFKA_SSL_KEY_CREDENTIALS
          value: "key_credentials"
        - name: KAFKA_SSL_TRUSTSTORE_FILENAME
          value: "kafka.truststore.jks"
        - name: KAFKA_SSL_TRUSTSTORE_CREDENTIALS
          value: "truststore_credentials"
        - name: KAFKA_SSL_CLIENT_AUTH
          value: "none"
        # Disable hostname verification ΓÇö our wildcard cert (*.service-virtualization.local)
        # does not cover multi-level hostnames like events.kafka.margin.service-virtualization.local.
        - name: KAFKA_SSL_ENDPOINT_IDENTIFICATION_ALGORITHM
          value: ""
        # =======================================================================
        # Single-node replication settings (service virtualization ΓÇö not HA)
        # =======================================================================
        - name: KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR
          value: "1"
        - name: KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR
          value: "1"
        - name: KAFKA_TRANSACTION_STATE_LOG_MIN_ISR
          value: "1"
        # Speeds up consumer group rebalance in single-broker setup
        - name: KAFKA_GROUP_INITIAL_REBALANCE_DELAY_MS
          value: "0"
        resources:
          limits:
            cpu: 1000m
            memory: 2Gi
          requests:
            cpu: 500m
            memory: 1Gi

        volumeMounts:
        - name: kafka-secrets-events
          mountPath: /etc/kafka/secrets
          readOnly: false
        livenessProbe:
          tcpSocket:
            port: 9092
          initialDelaySeconds: 60
          timeoutSeconds: 10
          periodSeconds: 30
          failureThreshold: 3
        readinessProbe:
          tcpSocket:
            port: 9092
          initialDelaySeconds: 45
          timeoutSeconds: 5
          periodSeconds: 10
          failureThreshold: 3
        startupProbe:
          tcpSocket:
            port: 9092
          initialDelaySeconds: 10
          timeoutSeconds: 10
          periodSeconds: 10
          failureThreshold: 30

      volumes:
      # emptyDir populated by the cert-converter init container.
      # Contains: kafka.keystore.jks, kafka.truststore.jks, and three credential files.
      - name: kafka-secrets-events
        emptyDir: {}
      # The team TLS secret ΓÇö read by the init container, not the main container.
      - name: margin-tls-secret
        secret:
          secretName: margin-tls-secret
          items:
            - key: tls.crt
              path: tls.crt
            - key: tls.key
              path: tls.key

    description: "Routes events.kafka.margin.service-virtualization.local:443 (TLS passthrough) to Kafka events pod port 9094"

  hosts:
  - "events.kafka.margin.service-virtualization.local"
  gateways:
  - istio-ingress/shared-gateway
  tls:
  - match:
    - port: 443
      sniHosts:
      - "events.kafka.margin.service-virtualization.local"
    route:
    - destination:
        host: margin-service-virtualization-kafka-events.margin.svc.cluster.local
        port:
          number: 9094
